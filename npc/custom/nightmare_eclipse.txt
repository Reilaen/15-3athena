function	script	ne_give_rewards	{
	.@party_id = getcharid(1);
	if (!.@party_id) return;

	.@current_map$ = strcharinfo(3);

	getpartymember(.@party_id, 0);
	copyarray(.@names$[0], $@partymembername$[0], $@partymembercount);

	getpartymember(.@party_id, 2);
	copyarray(.@aids[0], $@partymemberaid[0], $@partymembercount);

	.@count = $@partymembercount;
	for (.@i = 0; .@i < .@count; .@i++) {
		if (getmapxy(.@p_map$, .@x, .@y, UNITTYPE_PC, .@names$[.@i]) == 0) {
			if (.@p_map$ == .@current_map$) {
				getitem($@ne_cucrrency_id, 1, .@aids[.@i]);
				dispbottom2("0x00ff99", "A dark energy condenses in your hands... You received a Condensed Nightmare!", .@names$[.@i]);
			}
		}
	}
	return;
}
