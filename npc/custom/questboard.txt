	function	AddQuest	{
		.quest_count++;

		setd(".qb_" + .quest_count + "_name$", getarg(0));

		.@quest_type = getarg(1);
		setd(".qb_" + .quest_count + "_type", .@quest_type);

		setd(".qb_" + .quest_count + "_min_level", getarg(2));
		setd(".qb_" + .quest_count + "_max_level", getarg(3));
		explode(getd(".qb_" + .quest_count + "_reward_items"), getarg(4), ":");
		setd(".qb_" + .quest_count + "_random_item", getarg(5));
		setd(".qb_" + .quest_count + "_reward_zeny", getarg(6));
		setd(".qb_" + .quest_count + "_reward_nyan_points", getarg(7));
		setd(".qb_" + .quest_count + "_reward_event_points", getarg(8));
		setd(".qb_" + .quest_count + "_reward_base_exp", getarg(9));
		setd(".qb_" + .quest_count + "_reward_job_exp", getarg(10));
		explode(getd(".qb_" + .quest_count + "_collection_requirements"), getarg(11), ":");

		if(.@quest_type == 2) {
			for(.@i = 0; .@i < getarraysize(getd(".qb_" + .quest_count + "_collection_requirements")); .@i += 2) {
				.@mob_id = getd(".qb_" + .quest_count + "_collection_requirements[" + .@i + "]");

				if(inarray(.qb_mob_requirements, .@mob_id) == -1) {
					.qb_mob_requirements[getarraysize(.qb_mob_requirements)] = .@mob_id;
				}
			}
		}

		return;
	}

	function	ShowQuestInfo	{
		.@quest_id = getarg(0);
		mes("--------------------------------");
		mes("^0000FF" + getd(".qb_" + .@quest_id + "_name$") + "^000000");
		mes(" ");
		mes("[Requirement(s)]");

		for(.@i = 0; .@i < getarraysize(getd(".qb_"+ .@quest_id + "_collection_requirements")); .@i += 2) {
			.@requirement_id = getd(".qb_" + .@quest_id + "_collection_requirements[" + .@i + "]");
			.@requirement_amount = getd(".qb_" + .@quest_id + "_collection_requirements[" + (.@i + 1) + "]");

			if(getd(".qb_" + .@quest_id + "_type") != 2) {
				.@current_amount = countitem(.@requirement_id);
				.@requirement_name$ = "<ITEM>" + getitemname(.@requirement_id) + "<INFO>" + .@requirement_id + "</INFO></ITEM>";
			} else {
				.@current_amount = getd("qb_" + .@quest_id + "_" + .@requirement_id + "_kill_count");
				.@requirement_name$ = getmonsterinfo(.@requirement_id, MOB_NAME);
			}

			mes(((.@current_amount >= .@requirement_amount) ? "^00FF00" : "^FF0000") + .@requirement_name$ + " - " + .@current_amount + "/" + .@requirement_amount + "^000000");
		}
		mes(" ");
		mes("[Reward(s)]");
		.@reward_count = getarraysize(getd(".qb_" + .@quest_id + "_reward_items"));
		if(.@reward_count > 0) {
			if(getd(".qb_" + .@quest_id + "_random_item")) {
				mes("One of (chosen randomly):");
			}

			for(.@i = 0; .@i < .@reward_count; .@i += 2) {
				.@reward_id = getd(".qb_" + .@quest_id + "_reward_items[" + .@i + "]");
				mes("Item: ^0000FF" + getd(".qb_" + .@quest_id + "_reward_items[" + (.@i + 1) + "]") + " x <ITEM>" + getitemname(.@reward_id) + "<INFO>" + .@reward_id + "</INFO></ITEM>^000000");
			}
		}

		if(getd(".qb_" + .@quest_id + "_reward_zeny") != 0)
			mes("Zeny: ^0000FF" + getd(".qb_" + .@quest_id + "_reward_zeny") +  "^000000");

		if(getd(".qb_" + .@quest_id + "_reward_nyan_points") != 0)
			mes("Nyan Points: ^0000FF" + getd(".qb_" + .@quest_id + "_reward_nyan_points") + "^000000");

		if(getd(".qb_" + .@quest_id + "_reward_event_points") != 0)
			mes("Event Points: ^0000FF" + getd(".qb_" + .@quest_id + "_reward_event_points") + "^000000");

		if(getd(".qb_" + .@quest_id + "_reward_base_exp") != 0)
			mes("Base EXP: ^0000FF" + getd(".qb_" + .@quest_id + "_reward_base_exp") + "^000000");

		if(getd(".qb_" + .@quest_id + "_reward_job_exp") != 0)
			mes("Job EXP: ^0000FF" + getd(".qb_" + .@quest_id + "_reward_job_exp") + "^000000");

		mes("--------------------------------");

		return;
	}

	function	IncreaseKillCount	{
		.@mob_id = getarg(0);

		.@required_kills = getd("qb_mob_requirements_" + .@mob_id);
		.@current_kill_count = getd("qb_" + qb_current_quest_id + "_" + .@mob_id + "_kill_count") + 1;

		if (.@current_kill_count > .@required_kills) return;

		setd("qb_" + qb_current_quest_id + "_" + .@mob_id + "_kill_count", .@current_kill_count);
		dispbottom(getd(".qb_" + qb_current_quest_id + "_name$") + ": [" + getmonsterinfo(.@mob_id, MOB_NAME) + "] (" + .@current_kill_count + "/" + .@required_kills + ")");

		return;
	}

	function	AddNyanPoints	{
		.@account_id = getcharid(3);
		.@nyan_points = getarg(0);

		query_sql("INSERT INTO `vote` (`account_id`, `coins`) VALUES (" + .@account_id + ", " + .@nyan_points + ") ON DUPLICATE KEY UPDATE `coins` = `coins` + " + .@nyan_points);
		query_sql("INSERT INTO `cp_nyanpoint_log` (account_id, count, modified_id, time, type) VALUES (" + .@account_id + ", " + .@nyan_points + ", 0, NOW(), 'Questboard')");
	}

	function	AddEventPoints	{
		.@account_id = getcharid(3);
		.@event_points = getarg(0);

		query_sql("SELECT `userid` FROM `login` WHERE account_id = " + .@account_id, .@user_id$);
		query_sql("INSERT INTO `eventpoints` (`account_id`, `account_name`, `coins`) VALUES (" + .@account_id + ", '" + escape_sql(.@user_id$) + "', " + .@event_points + ") ON DUPLICATE KEY UPDATE `coins` = `coins` + " + .@event_points);
		query_sql("INSERT INTO `cp_eventpoint_log` (account_id, count, modified_id, time, type) VALUES (" + .@account_id + ", " + .@event_points + ", 0, NOW(), 'Questboard')");
	}
