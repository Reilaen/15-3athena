			mes("[Modsognir]");

			next;

		.ownedAmt = countitem(.currency_item);

			mes("[Modsognir]");

			close;
		}

		.@selected_amt = 1;
		if(.ownedAmt > 1) {

			mes("[Modsognir]");
			mes("How many ^0000FF" + getitemname(.currency_item) + "^000000 do you want to trade?");
			next;
			.@validation = input(.@selected_amt, 1, 200);
			if(.@validation != 0) {
				mes("[Modsognir]");
				mes((.@validation > 0) ? "That is too much to process for me at once.." : "That does not make sense. Did you really mean that?");
				next;
				goto(SelectAmt);
			}
		}

		.@left_today = .reward_limit - #re_daily_blessings;
		.@primary_amt = min(.@selected_amt, .@left_today);
		.@secondary_amt = .@selected_amt - .@primary_amt;

		if (.@secondary_amt > 0 && .@left_today > 0) {
			mes("[Modsognir]");
			mes("That is quite an amount! However, I only have ^0000FF" + .@left_today + " " + getitemname(.reward) + "^000000 left in stock today.");
			mes("For the remaining " + .@secondary_amt + " items, I will give you other useful materials from my supplies instead. Is that alright with you?");
			next;
			if(select("Yes, that sounds fair.:No, let's cancel.") == 2) goto(Cancel);
		}

		if(.@primary_amt > 0) {
			if(!checkweight(.reward, .@primary_amt)) {
				goto(NotEnoughSpace);
			}

			getitem(.reward, .@primary_amt);
			delitem(.currency_item, .@primary_amt);
			#re_daily_blessings += .@primary_amt;
		}

		for(.@i = 0; .@i < .@secondary_amt; .@i++) {
			.@rnd_idx = rand(1, .seconday_reward_count);
			.@reward_var$ = ".reward_ids_" + .@rnd_idx;
			.@reward_amt_var$ = ".reward_amt_" + .@rnd_idx;

			if(!checkweight2(getd(.@reward_var$), getd(.@reward_amt_var$))) {
				goto(NotEnoughSpace);
			}

			for(.@y = 0; .@y < getarraysize(getd(.@reward_var$)); .@y++) {
				getitem(getd(.@reward_var$ + "[" + .@y + "]"), getd(.@reward_amt_var$ + "[" + .@y + "]"));
			}
			delitem(.currency_item, 1);
		}

		mes("[Modsognir]");
		mes("Enjoy! You have really earned it.");
		close;

		}

			mes("^0000FF- " + getd(".mob_map_" + .hint_mob_ids[.@i] + "$") + "^000000");
		}

		close;

			.@mobid = atoi(.@atcmd_parameters$[0]);

			if (.@mobid == 0) {
				dispbottom("Usage: @findthief <MobID> (z.B. @findthief 1002)");
				end;
			}

			.@gid = getd(".mob_gids_" + .@mobid);
			if (!unitexists(.@gid)) {
				dispbottom("Error: No Mineral Thief with the ID " + .@mobid + " currently exists.");
				end;
			}

			.@map$ = getd(".mob_map_" + .@mobid + "$");
			.@mob = getunitdata(.@gid, .@mob);
			.@x = .@mob[UMOB_X];
			.@y = .@mob[UMOB_Y];

			.@hp = .@mob[UMOB_HP];
			.@maxhp = .@mob[UMOB_MAXHP];

			dispbottom("Mob ID: " + .@mobid);
			dispbottom("Position: " + .@map$ + " (" + .@x + " / " + .@y + ")");
			end;

		OnSpawnMob:
			.@mobid = getarg(0);
			.@idx   = rand(getarraysize(getd(".maps_" + .@mobid + "$")));
			.@map$  = getd(".maps_" + .@mobid + "$[" + .@idx + "]");

			setd(".mob_map_" + .@mobid + "$", .@map$);
			monster(.@map$, -1, -1, "Mineral Thief", .@mobid, 1, strnpcinfo(0) + "::OnThiefDead");
			setd(".mob_gids_" + .@mobid, $@mobid[0]);
			return;

			dispbottom("The Mineral Thief escaped to somewhere else..");

			end;

				.@mobid = .mobs[.@i];
                .@gid = getd(".mob_gids_" + .@mobid);

				if (unitexists(.@gid)) {
					 getunitdata(.@gid, .@mob);
					.@hp = .@mob[UMOB_HP];
					.@maxhp = .@mob[UMOB_MAXHP];

					if (.@hp == .@maxhp) {
						killmonster(getd(".mob_map_" + .@mobid + "$"), strnpcinfo(0) + "::OnThiefDead");
						callsub(OnSpawnMob, .@mobid);
					}
				}
			}
