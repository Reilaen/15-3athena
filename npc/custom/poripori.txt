		mapannounce('map$, "[Pori Pori Nya] The Pori Pori Challenge has started! You have 1 Minute to get ready!", bc_map, "0x00ff99");
		initnpctimer;
		end;

		mapwarp('map$, $@pori_return_map$, $@pori_return_coords[0], $@pori_return_coords[1]);
		instance_destroy();
		end;

			stopnpctimer;

	        for (.@i = 0; .@i < 20; .@i += 2) {
	            if ($pori_top_ten$[.@i] == "" || .@finish_time < atoi($pori_top_ten$[.@i + 1])) {
					for (.@j = 16; .@j >= .@i; .@j -= 2) {
			            $pori_top_ten$[.@j + 2] = $pori_top_ten$[.@j];
			            $pori_top_ten$[.@j + 3] = $pori_top_ten$[.@j + 1];
			        }

			        $pori_top_ten$[.@i + 1] = "" + .@finish_time;

	                announce("[Pori Pori Nya] New record! The party [" + .@party_name$ + "] took place " + ((.@i/2)+1) + "!", bc_map, "0xFFFF00");
	                break;
	            }
	        }

			.@finish_time_str$ = callfunc("F_FormatMilliseconds", .@finish_time);
			.@body$ = "You beat the final level in " + .@finish_time_str$ + "!\r\n" +
						"That is really impressive.\r\n" +
						"Enjoy your price!";
