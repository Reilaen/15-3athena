-	script	Castle_Raid	-1,{
	OnInit:
		if(strnpcinfo(4) == "") end;

		bindatcmd("startraid",strnpcinfo(0) + "::OnStartRaid", 99, 99);
		bindatcmd("endraid",strnpcinfo(0) + "::OnEndRaid", 99, 99);
		.castle_map$ = strnpcinfo(4);

		// WoE FE
		if (.castle_map$ == "aldeg_cas01") setarray .emproom[0],216,23;
		else if (.castle_map$ == "aldeg_cas02") setarray .emproom[0],213,23;
		else if (.castle_map$ == "aldeg_cas03") setarray .emproom[0],205,31;
		else if (.castle_map$ == "aldeg_cas04") setarray .emproom[0],36,217;
		else if (.castle_map$ == "aldeg_cas05") setarray .emproom[0],27,101;
		else if (.castle_map$ == "gefg_cas01") setarray .emproom[0],197,181;
		else if (.castle_map$ == "gefg_cas02") setarray .emproom[0],176,178;
		else if (.castle_map$ == "gefg_cas03") setarray .emproom[0],244,166;
		else if (.castle_map$ == "gefg_cas04") setarray .emproom[0],174,177;
		else if (.castle_map$ == "gefg_cas05") setarray .emproom[0],194,184;
		else if (.castle_map$ == "payg_cas01") setarray .emproom[0],139,139;
		else if (.castle_map$ == "payg_cas02") setarray .emproom[0],38,25;
		else if (.castle_map$ == "payg_cas03") setarray .emproom[0],269,265;
		else if (.castle_map$ == "payg_cas04") setarray .emproom[0],270,28;
		else if (.castle_map$ == "payg_cas05") setarray .emproom[0],30,30;
		else if (.castle_map$ == "prtg_cas01") setarray .emproom[0],197,197;
		else if (.castle_map$ == "prtg_cas02") setarray .emproom[0],157,174;
		else if (.castle_map$ == "prtg_cas03") setarray .emproom[0],16,220;
		else if (.castle_map$ == "prtg_cas04") setarray .emproom[0],291,14;
		else if (.castle_map$ == "prtg_cas05") setarray .emproom[0],266,266;
		// WoE SE
		else if (.castle_map$ == "arug_cas01") setarray .emproom[0],87,219;
		else if (.castle_map$ == "arug_cas02") setarray .emproom[0],89,256;
		else if (.castle_map$ == "arug_cas03") setarray .emproom[0],141,293;
		else if (.castle_map$ == "arug_cas04") setarray .emproom[0],141,293;
		else if (.castle_map$ == "arug_cas05") setarray .emproom[0],141,293;
		else if (.castle_map$ == "schg_cas01") setarray .emproom[0],120,272;
		else if (.castle_map$ == "schg_cas02") setarray .emproom[0],162,193;
		else if (.castle_map$ == "schg_cas03") setarray .emproom[0],338,202;
		else if (.castle_map$ == "schg_cas04") setarray .emproom[0],120,272;
		else if (.castle_map$ == "schg_cas05") setarray .emproom[0],120,272;
		// WoE TE
		else if (.castle_map$ == "te_aldecas1") setarray .emproom[0],216,23;
		else if (.castle_map$ == "te_aldecas2") setarray .emproom[0],213,23;
		else if (.castle_map$ == "te_aldecas3") setarray .emproom[0],205,31;
		else if (.castle_map$ == "te_aldecas4") setarray .emproom[0],36,217;
		else if (.castle_map$ == "te_aldecas5") setarray .emproom[0],27,101;
		else if (.castle_map$ == "te_prtcas01") setarray .emproom[0],197,197;
		else if (.castle_map$ == "te_prtcas02") setarray .emproom[0],157,174;
		else if (.castle_map$ == "te_prtcas03") setarray .emproom[0],16,220;
		else if (.castle_map$ == "te_prtcas04") setarray .emproom[0],291,14;
		else if (.castle_map$ == "te_prtcas05") setarray .emproom[0],266,266;
	end;

	OnStartRaid:
	OnWed1955:
		.owning_guild_id = getcastledata(.castle_map$, 1);
		if(.owning_guild_id == 0) end;

		initnpctimer;
		announce("A Raid Group is forming around [" + getcastlename(.castle_map$) + "] Castle! Members of [" + getguildName(.owning_guild_id) + "] Guild, you have 5 minutes to gather to defend it!", bc_all, 0xFF0000);
		end;

	OnTimer300000:
		.event_active = 1;
		setmapflag(.castle_map$, MF_LOADEVENT);
		maprespawnguildid(.castle_map$, .owning_guild_id, 2);

		.main_phase_kills = 0;
		.captain_kills = 0;
		.member_count = getmapguildusers(.castle_map$, .owning_guild_id);
		.tier = callsub(S_CalcBaselineTier, .owning_guild_id, 1, .member_count);

		.captains = $@captains;
		.scouts = $@scouts + .member_count * $@scouts_per_player;

		announce("A Raid Group has started attacking [" + getcastlename(.castle_map$) + "] castle!", bc_all, 0xFF0000);
		mapannounce(.castle_map$, "The [" + getcastlename(.castle_map$) + "] Castle will be conquered if the Raiders are not defeated within 30 minutes!", bc_map, 0xFF0000);
		mapannounce(.castle_map$, .scouts + " Raid Scouts have appeared! Defeat them before the main force arrives in 5 minutes!", bc_map, 0xFF0000);
		callsub(S_SpawnScouts, .scouts, .tier);
		end;

	OnTimer600000:
		mapannounce(.castle_map$, "The Raiders main forces have arrived! The Raid Leader dispatches the first of " + .captains + " Raid Captains to break your defenses!", bc_map, 0xFF0000);
		if(mobcount(.castle_map$, strnpcinfo(0) + "::OnRaidScoutKilled") > 0) {
			.captains += $@captain_advantage;
		}

		mapannounce(.castle_map$, "Find and defeat the " + .captains + " Raid Captain!", bc_map, 0xFF0000);

		callsub(S_SpawnRaiders, $@raiders + .member_count * $@raiders_per_player, .tier);
		callsub(S_SpawnCaptains, 1, .tier);
		end;

	OnTimer2100000:
	OnDefendFailed:
		announce("The [" + getguildName(.owning_guild_id) + "] Guild lost [" + getcastlename(.castle_map$) + "] Castle to the Raid Group!", bc_all, 0xFF0000);
		for (.@i = 1; .@i <= 17; .@i++) {
			setcastledata(.castle_map$, .@i, 0);
		}

		maprespawnguildid(.castle_map$, .owning_guild_id, 1);
		callsub(S_EndEvent);
		end;

	OnRaidScoutKilled:
		.main_phase_kills++;

		if (mobcount(.castle_map$, strnpcinfo(0) + "::OnRaidScoutKilled") == 0) {
			mapannounce(.castle_map$, "All Raid Scouts have been defeated! Use the earned time to prepare.", bc_map, 0xFF0000);
			.@elapsed_ms = getnpctimer(0) - 300000;

			if (.@elapsed_ms <= $@scout_tier3_time && .tier < 3) {
				.tier = 3;
			} else if (.@elapsed_ms <= $@scout_tier2_time && .tier < 2) {
				.tier = 2;
			}
			end;
		}

		mapannounce(.castle_map$, "A Raid Scout has been defeated! (" + mobcount(.castle_map$, strnpcinfo(0) + "::OnRaidScoutKilled") + " / " + .scouts + " remaining)", bc_map, 0xFF0000);
		end;

	OnRaidMemberKilled:
		.main_phase_kills++;

		.tier = callsub(S_CalcTierByKills, .tier);
		callsub(S_SpawnRaiders, 1, .tier);
		end;

	OnRaidCaptainKilled:
		.main_phase_kills++;
		.captain_kills++;

		.tier = callsub(S_CalcTierByKills, .tier);
		if(.captain_kills < .captains) {
			mapannounce(.castle_map$, "A Raid Captain has fallen! (" + (.captains - .captain_kills) +  " / " + .captains + " remaining)", bc_map, 0xFF0000);
			mapannounce(.castle_map$, "But there is no time to rest! Another Raid Captain just broke through the defenses!", bc_map, 0xFF0000);
			callsub(S_SpawnCaptains, 1, .tier);
			end;
		}

		mapannounce(.castle_map$, "Angered by the miserable failure of his Raid Captains, the Raid Leader has burst in the Emperium Room!", bc_map, 0xFF0000);
		callsub(S_SpawnLeader, .tier);
		end;

	OnEndRaid:
	OnRaidLeaderKilled:
		callsub(S_EndEvent);
		announce("The [" + getguildName(.owning_guild_id) + "] Guild successfully defended [" + getcastlename(.castle_map$) + "] Castle against the Raid Group!", bc_all, 0xFF0000);
		end;

	OnPCDieEvent:
		if(!.event_active || strcharinfo(3) != .castle_map$ || getmapguildusers(.castle_map$, .owning_guild_id) > 0) end;
		goto(OnDefendFailed);

	OnPCLoadMapEvent:
		if (strcharinfo(3) != .castle_map$ || getgmlevel() > 0) end;
		sleep2(500);
		dispbottom("The [" + getcastlename(.castle_map$) + "] Castle is surrounded by a magical barrier, which makes it impossible to enter.");
		warp("SavePoint", 0, 0);
		end;

	S_EndEvent:
		stopnpctimer;
		callsub(S_ClearMonsters);
		removemapflag(.castle_map$, MF_LOADEVENT);
		.event_active = 0;
		return;

		S_CalcBaselineTier:
		    .@guild_id = getarg(0);
		    .@min_tier = getarg(1, 1);
		    .@member_count = getarg(2, 1);
		    .@total_power = 0;

		    getmapunits(BL_PC, .castle_map$, .@players$[0]);
		    for (.@i = 0; .@i < getarraysize(.@players$); .@i++ ) {
		        if (getcharid(2, .@players$[.@i]) == .@guild_id) {
		            .@member_count++;

		            .@atk = readparam(bAtk, .@players[.@i]) + readparam(bAtk2, .@players[.@i]);
		            .@matk = readparam(bMatk, .@players[.@i]) + readparam(bMatk2, .@players[.@i]);
		            .@total_power += (.@atk + .@matk);
		        }
		    }

		    .@tier_by_members = 1;
		    if (.@member_count >= $@member_count_tier3_threshold) {
				.@tier_by_members = 3;
		    } else if (.@member_count >= $@member_count_tier2_threshold) {
				.@tier_by_members = 2;
			}

		    .@tier_by_offense = 1;
		    if (.@total_power >= $@high_offense_threshold) {
				.@tier_by_offense = 3;
			} else if (.@total_power >= $@medium_offense_threshold) {
				.@tier_by_offense = 2;
			}

		    .@calc_tier = .@tier_by_members;
		    if (.@tier_by_offense > .@calc_tier) {
		        .@calc_tier = .@tier_by_offense;
		    }

		    if (.@min_tier > .@calc_tier) {
				return .@min_tier;
			}

		return .@calc_tier;

	S_CalcTierByKills:
		.@min_tier = getarg(0, 1);
		.@calc_tier = 1;

		.@elapsed_ms = getnpctimer(0) - 300000;
	    if (.@elapsed_ms > 0) {
	        .@kills_per_minute = (.main_phase_kills * 60000) / .@elapsed_ms;

	        if (.@kills_per_minute >= $@kpm_tier3_threshold) {
	            .@calc_tier = 3;
	        } else if (.@kills_per_minute >= $@kpm_tier2_threshold) {
	            .@calc_tier = 2;
	        }
	    }

		if (.@min_tier > .@calc_tier) return .@min_tier;

		return .@calc_tier;

	S_SpawnScouts:
		callsub(S_SpawnBasicRaiders, getarg(0), getarg(1), "Raid Scout", "OnRaidScoutKilled");
		return;

	S_SpawnRaiders:
		callsub(S_SpawnBasicRaiders, getarg(0), getarg(1), "Raider", "OnRaidMemberKilled");
		return;

	S_SpawnBasicRaiders:
		.@name$ = getarg(2);
		.@label$ = getarg(3);

		switch(getarg(1)) {
			case 1:
				copyarray(.@raider_ids[0], $@low_tier_raiders[0], getarraysize($@low_tier_raiders));
				break;
			case 2:
				copyarray(.@raider_ids[0], $@medium_tier_raiders[0], getarraysize($@medium_tier_raiders));
				break;
			case 3:
				copyarray(.@raider_ids[0], $@high_tier_raiders[0], getarraysize($@high_tier_raiders));
				break;
		}

		.@raider_amount = getarg(0);
		for(.@i = 0; .@i < .@raider_amount; .@i++) {
			monster(.castle_map$, 0, 0, .@name$, .@raider_ids[rand(0, getarraysize(.@raider_ids) - 1)], 1, strnpcinfo(0) + "::" + .@label$);
		}

		return;

	S_SpawnCaptains:
		switch(getarg(1)) {
			case 1:
				copyarray(.@captain_ids[0], $@low_tier_captains[0], getarraysize($@low_tier_captains));
				break;
			case 2:
				copyarray(.@captain_ids[0], $@medium_tier_captains[0], getarraysize($@medium_tier_captains));
				break;
			case 3:
				copyarray(.@captain_ids[0], $@high_tier_captains[0], getarraysize($@high_tier_captains));
				break;
		}

		.@captain_amount = getarg(0);
		for(.@i = 0; .@i < .@captain_amount; .@i++) {
			monster(.castle_map$, 0, 0, "Raid Captain", .@captain_ids[rand(0, getarraysize(.@captain_ids) - 1)], 1, strnpcinfo(0) + "::OnRaidCaptainKilled");
		}
		return;

	S_SpawnLeader:
		switch(getarg(0)) {
			case 1:
				copyarray(.@leader_ids[0], $@low_tier_leaders[0], getarraysize($@low_tier_leaders));
				break;
			case 2:
				copyarray(.@leader_ids[0], $@medium_tier_leaders[0], getarraysize($@medium_tier_leaders));
				break;
			case 3:
				copyarray(.@leader_ids[0], $@high_tier_leaders[0], getarraysize($@high_tier_leaders));
				break;
		}

		monster(.castle_map$, .emproom[0], .emproom[1], "Raid Leader", .@leader_ids[rand(0, getarraysize(.@leader_ids) - 1)], 1, strnpcinfo(0) + "::OnRaidLeaderKilled");
		return;

	S_ClearMonsters:
			killmonster(.castle_map$, strnpcinfo(0) + "::OnRaidScoutKilled");
			killmonster(.castle_map$, strnpcinfo(0) + "::OnRaidMemberKilled");
			killmonster(.castle_map$, strnpcinfo(0) + "::OnRaidCaptainKilled");
			killmonster(.castle_map$, strnpcinfo(0) + "::OnRaidLeaderKilled");
			return;
}
